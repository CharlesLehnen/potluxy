-- Enable pgcrypto for UUID generation
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

---------------------------
-- 1. EVENTS TABLE
---------------------------
CREATE TABLE events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_name TEXT NOT NULL,
    slug TEXT UNIQUE NOT NULL,
    created_by TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    event_description TEXT,
    event_date DATE,
    event_time TIME
);

---------------------------
-- 2. EVENT ITEMS TABLE
---------------------------
CREATE TABLE event_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
    item_name TEXT NOT NULL,
    quantity_required INT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    item_description TEXT,
    UNIQUE(event_id, item_name)
);

---------------------------
-- 3. EVENT_USERS TABLE (Event-Specific Profiles)
---------------------------
CREATE TABLE event_users (
    id UUID PRIMARY KEY,
    event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
    username TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    UNIQUE(event_id, username)
);

---------------------------
-- 4. COMMITMENTS TABLE (User Commitments)
---------------------------
CREATE TABLE commitments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
    user_id UUID REFERENCES event_users(id) ON DELETE SET NULL,
    item_id UUID NOT NULL REFERENCES event_items(id) ON DELETE CASCADE,
    quantity INT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    UNIQUE(event_id, user_id, item_id)
);

---------------------------
-- Enable Row Level Security (RLS)
---------------------------
ALTER TABLE events ENABLE ROW LEVEL SECURITY;
ALTER TABLE event_items ENABLE ROW LEVEL SECURITY;
ALTER TABLE event_users ENABLE ROW LEVEL SECURITY;
ALTER TABLE commitments ENABLE ROW LEVEL SECURITY;

---------------------------
-- RLS Policies
---------------------------

-- EVENTS: Public SELECT and INSERT are allowed.
CREATE POLICY "Public select on events" ON events FOR SELECT USING (true);
CREATE POLICY "Public insert on events" ON events FOR INSERT WITH CHECK (true);

-- Only allow UPDATE/DELETE if the current auth user is the creator.
CREATE POLICY "Event update" ON events FOR UPDATE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = events.id
        AND event_users.username = events.created_by
    )
);
CREATE POLICY "Event delete" ON events FOR DELETE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = events.id
        AND event_users.username = events.created_by
    )
);

-- EVENT ITEMS: Public SELECT; for INSERT/UPDATE/DELETE, only allow if current user is the event creator.
CREATE POLICY "Event items select" ON event_items FOR SELECT USING (
    EXISTS (SELECT 1 FROM events WHERE events.id = event_items.event_id)
);
CREATE POLICY "Event items insert" ON event_items FOR INSERT
WITH CHECK (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = event_items.event_id
        AND event_users.username = (SELECT created_by FROM events WHERE events.id = event_items.event_id)
    )
);
CREATE POLICY "Event items update" ON event_items FOR UPDATE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = event_items.event_id
        AND event_users.username = (SELECT created_by FROM events WHERE events.id = event_items.event_id)
    )
);
CREATE POLICY "Event items delete" ON event_items FOR DELETE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = event_items.event_id
        AND event_users.username = (SELECT created_by FROM events WHERE events.id = event_items.event_id)
    )
);

CREATE POLICY "Event users select" ON event_users FOR SELECT
  USING (true);

CREATE POLICY "Event users insert" ON event_users FOR INSERT
WITH CHECK (id = auth.uid());

-- COMMITMENTS: Users may only access their own commitments.
CREATE POLICY "Commitments select" ON commitments FOR SELECT
  USING (true);

CREATE POLICY "Commitments insert" ON commitments FOR INSERT
WITH CHECK (user_id = auth.uid());

CREATE POLICY "Commitments update" ON commitments FOR UPDATE
USING (user_id = auth.uid());

CREATE POLICY "Commitments delete" ON commitments FOR DELETE
USING (user_id = auth.uid());

---------------------------
-- Additional Constraints
---------------------------

-- Prevent overcommitments on INSERT
CREATE POLICY "Prevent overcommitments on insert"
ON commitments
FOR INSERT
WITH CHECK (
    quantity <= (
        SELECT quantity_required - COALESCE((
            SELECT SUM(quantity)
            FROM commitments
            WHERE item_id = commitments.item_id
        ), 0)
        FROM event_items
        WHERE id = commitments.item_id
    )
);

-- Prevent overcommitments on UPDATE
CREATE POLICY "Prevent overcommitments on update"
ON commitments
FOR UPDATE
WITH CHECK (
    quantity <= (
        SELECT quantity_required - COALESCE((
            SELECT SUM(quantity)
            FROM commitments
            WHERE item_id = commitments.item_id
        ), 0)
        FROM event_items
        WHERE id = commitments.item_id
    )
);


-- Prevent reducing quantity below existing commitments
CREATE OR REPLACE FUNCTION prevent_reduce_below_commitment()
RETURNS TRIGGER AS $$
DECLARE
    total_committed INT;
BEGIN
    SELECT COALESCE(SUM(quantity), 0)
    INTO total_committed
    FROM commitments
    WHERE item_id = NEW.id;

    IF NEW.quantity_required < total_committed THEN
        RAISE EXCEPTION 'Cannot reduce required quantity below existing commitments (%).', total_committed;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger to enforce the above constraint
CREATE TRIGGER prevent_reduce_below_commitment_trigger
BEFORE UPDATE ON event_items
FOR EACH ROW
EXECUTE FUNCTION prevent_reduce_below_commitment();

// Function to initialize Supabase client by securely fetching credentials from the Edge Function
async function initializeSupabaseClient() {
  const EDGE_FUNCTION_URL = "https://your-project-ref.supabase.co/functions/v1/getKeys";
  try {
    const response = await fetch(EDGE_FUNCTION_URL);
    if (!response.ok) throw new Error("Failed to fetch Supabase keys");
    const { SUPABASE_URL, SUPABASE_ANON_KEY } = await response.json();
    supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);
    console.log("Supabase initialized securely.");
  } catch (error) {
    console.error("Error initializing Supabase:", error);
  }
}

function constructEmail(username, slug) {
  return `${username}.${slug}@mailinator.com`;
}

function generateSlug(eventName) {
  const uniqueId = Math.random().toString(36).substring(2, 10);
  return eventName.toLowerCase().replace(/\s+/g, '-') + '-' + uniqueId;
}

async function createEvent(e) {
  e.preventDefault();
  const eventName = document.getElementById('eventName').value.trim();
  const eventDescription = document.getElementById('eventDescription').value.trim();
  const eventDate = document.getElementById('eventDate').value;
  const eventTime = document.getElementById('eventTime').value;
  const username = document.getElementById('username').value.trim();
  const password = document.getElementById('password').value.trim();
  if (!eventName || !username) {
    alert('Event name and username are required.');
    return;
  }
  const slug = generateSlug(eventName);
  const email = constructEmail(username, slug);
  
  // Sign up via Supabase Auth.
  const { data: authData, error: signUpError } = await supabase.auth.signUp({
    email,
    password: password || 'no-password'
  });
  if (signUpError) {
    alert(`Error signing up: ${signUpError.message}`);
    return;
  }
  
  const { data: sessionData, error: sessionError } = await supabase.auth.getSession();
  if (sessionError || !sessionData.session) {
    alert('Error: User not fully authenticated after signup.');
    return;
  }
  const userId = sessionData.session.user.id;
  
  // Create the event.
  const { data: eventData, error: eventError } = await supabase
    .from('events')
    .insert([{
      event_name: eventName,
      slug,
      created_by: username,
      event_description: eventDescription || null,
      event_date: eventDate || null,
      event_time: eventTime || null
    }])
    .select()
    .single();
  if (eventError) {
    alert(`Error creating event: ${eventError.message}`);
    return;
  }
  
  const eventId = eventData.id;
  
  // Create event-specific profile.
  const { data: profileData, error: profileError } = await supabase
    .from('event_users')
    .insert([{ id: userId, event_id: eventId, username }])
    .select()
    .single();
  if (profileError) {
    alert(`Error creating event user: ${profileError.message}`);
    return;
  }
  currentUser = profileData;
  
  alert(`Event created successfully! Share this link: ${location.origin}/#/event/${slug}`);
  window.location.hash = `#/event/${slug}`;
  loadEvent(slug);
}

// ----------------------
// 2. Event Loading & Login (Per Event)
// ----------------------
async function loadEvent(slug) {
  const { data: eventData, error: eventError } = await supabase
    .from('events')
    .select('*, event_items(*)')
    .eq('slug', slug)
    .single();
  if (eventError || !eventData) {
    alert('Event not found.');
    renderCreateEventForm();
    return;
  }
  currentEvent = eventData;
  currentEvent.items = eventData.event_items || [];
  
  // Check Supabase Auth session.
  const { data: { session } } = await supabase.auth.getSession();
  if (!session) {
    renderLoginForm(slug);
  } else {
    const { data: profileData } = await supabase
      .from('event_users')
      .select('*')
      .eq('id', session.user.id)
      .eq('event_id', currentEvent.id)
      .single();
    if (!profileData) {
      alert("No event-specific profile found; please sign up for this event.");
      renderLoginForm(slug);
    } else {
      currentUser = profileData;
      renderEventDashboard();
    }
  }
}

  app.innerHTML = `

    <p>If you don't have an account for this event, signing up will create one.</p>
  `;
  document.getElementById('loginForm').addEventListener('submit', (e) => handleLogin(e, slug));
}

async function handleLogin(e, slug) {
  e.preventDefault();
  const username = document.getElementById('username').value.trim();
  const password = document.getElementById('password').value.trim();
  const email = constructEmail(username, slug);
  
  // Attempt to sign in
  let { data: signInData, error: signInError } = await supabase.auth.signInWithPassword({
    email, 
    password: password || 'no-password'
  });
  if (signInError) {
    if (signInError.message.includes("Invalid login credentials")) {
      const { data: signUpData, error: signUpError } = await supabase.auth.signUp({
        email, 
        password: password || 'no-password'
      });
      if (signUpError) {
        alert(`Error signing up: ${signUpError.message}`);
        return;
      }
      signInData = signUpData;
    } else {
      alert(`Error signing in: ${signInError.message}`);
      return;
    }
  }
  
  // Fetch (or create) the event-specific profile.
  const { data: profileData } = await supabase
    .from('event_users')
    .select('*')
    .eq('id', signInData.session.user.id)
    .eq('event_id', currentEvent.id)
    .single();
  if (!profileData) {
    const { data: newProfile, error: newProfileError } = await supabase
      .from('event_users')
      .insert([{ id: signInData.session.user.id, event_id: currentEvent.id, username }])
      .select()
      .single();
    if (newProfileError) {
      alert(`Error creating event user: ${newProfileError.message}`);
      return;
    }
    currentUser = newProfile;
  } else {
    currentUser = profileData;
  }
  renderEventDashboard();
}

function renderViewItems() {
  if (!currentEvent.items || currentEvent.items.length === 0) {
    return `<p>No items listed for this event yet.</p>`;
  }
  return currentEvent.items.map(item => `
    <div class="item-card">
      <p>${item.item_name} - ${item.quantity_required} needed</p>
      ${item.item_description ? `<p><em>${item.item_description}</em></p>` : ''}
    </div>
  `).join('');
}

async function renderCommitmentForm() {
  const { data: allCommitments } = await supabase
    .from('commitments')
    .select('*')
    .eq('event_id', currentEvent.id);
  const commitmentTotals = {};
  const userCommitments = {};
  if (allCommitments) {
    allCommitments.forEach(commit => {
      commitmentTotals[commit.item_id] = (commitmentTotals[commit.item_id] || 0) + commit.quantity;
      if (currentUser && commit.user_id === currentUser.id) {
        userCommitments[commit.item_id] = commit.quantity;
      }
    });
  }
  return currentEvent.items.map(item => {
    const totalCommitted = commitmentTotals[item.id] || 0;
    const yourCommitment = (userCommitments[item.id] === undefined) ? 0 : userCommitments[item.id];
    const remaining = item.quantity_required - totalCommitted;
    return `
      <div class="commitment-card">
        <strong style="text-align: center; display: block;">${item.item_name}</strong>
        <br><br>
        <div>Required: ${item.quantity_required}</div>
        <div>Total Committed: ${totalCommitted}</div>
        <div>Your Commitment: ${yourCommitment}</div>
        <div>
            <input type="number" id="commit-${item.id}" value="${yourCommitment}" min="0">
            Remaining: ${remaining}
            <button onclick="commitItem('${item.id}')">Commit</button>
        </div>
      </div>
    `;
  }).join('');
}

async function renderCommitmentBreakdown() {
  const { data: usersData } = await supabase
    .from('event_users')
    .select('id, username')
    .eq('event_id', currentEvent.id);
  const userLookup = {};
  if (usersData) {
    usersData.forEach(user => {
      if (user && user.username) {
        userLookup[user.id] = user.username;
      }
    });
  }
  const { data: breakdownData, error } = await supabase
    .from('commitments')
    .select('quantity, item_id, user_id')
    .eq('event_id', currentEvent.id);
  if (error) {
    return `<p>Error loading commitment breakdown.</p>`;
  }
  const breakdownByItem = {};
  breakdownData.forEach(row => {
    if (!row) return;
    const itemId = row.item_id;
    const username = row.user_id ? (userLookup[row.user_id] || "Unknown") : "Unknown";
    if (!breakdownByItem[itemId]) breakdownByItem[itemId] = [];
    breakdownByItem[itemId].push(`${username}: ${row.quantity}`);
  });
  let html = '<h4>Commitment Breakdown</h4>';
  currentEvent.items.forEach(item => {
    if (breakdownByItem[item.id]) {
      html += `<p><strong>${item.item_name}:</strong> ${breakdownByItem[item.id].join(', ')}</p>`;
    }
  });
  return html;
}

async function handleAddItem(e) {
  e.preventDefault();
  const newItemName = document.getElementById('newItemName').value.trim();
  const newItemQuantity = parseInt(document.getElementById('newItemQuantity').value);
  const newItemDescription = document.getElementById('newItemDescription').value.trim();
  if (!newItemName || newItemQuantity <= 0) {
    alert("Please provide a valid item name and quantity.");
    return;
  }
  const insertData = {
    event_id: currentEvent.id,
    item_name: newItemName,
    quantity_required: newItemQuantity
  };
  if (newItemDescription) {
    insertData.item_description = newItemDescription;
  }
  const { error } = await supabase
    .from('event_items')
    .insert([insertData]);
  if (error) {
    alert("Error adding item: " + error.message);
  } else {
    alert("Item added!");
    loadEvent(currentEvent.slug);
  }
}

async function updateItem(itemId) {
  if (currentUser?.username !== currentEvent.created_by) {
    alert("Only the event creator can update items.");
    return;
  }
  const newItemName = document.getElementById(`edit-item-${itemId}`).value.trim();
  const newQuantity = parseInt(document.getElementById(`edit-qty-${itemId}`).value);
  const newDescription = document.getElementById(`edit-desc-${itemId}`)?.value.trim();
  if (!newItemName || newQuantity <= 0) {
    alert("Please provide valid values.");
    return;
  }
  const { data: commitmentsData } = await supabase
    .from('commitments')
    .select('quantity')
    .eq('item_id', itemId);
  const totalCommitted = commitmentsData?.reduce((sum, c) => sum + c.quantity, 0) || 0;
  if (newQuantity < totalCommitted) {
    alert(`Cannot reduce required quantity to ${newQuantity}. ${totalCommitted} already committed.`);
    return;
  }
  const updateData = { item_name: newItemName, quantity_required: newQuantity };
  if (newDescription !== undefined) {
    updateData.item_description = newDescription;
  }
  const { error } = await supabase
    .from('event_items')
    .update(updateData)
    .eq('id', itemId);
  if (error) {
    alert(`Error updating item: ${error.message}`);
  } else {
    alert("Item updated successfully!");
    loadEvent(currentEvent.slug);
  }
}

async function deleteItem(itemId) {
  if (currentUser?.username !== currentEvent.created_by) {
    alert("Only the event creator can delete items.");
    return;
  }
  if (!confirm("Are you sure you want to delete this item? This cannot be undone.")) {
    return;
  }
  const { error } = await supabase
    .from('event_items')
    .delete()
    .eq('id', itemId);
  if (error) {
    alert(`Error deleting item: ${error.message}`);
  } else {
    alert("Item deleted successfully!");
    loadEvent(currentEvent.slug);
  }
}

async function commitItem(itemId) {
  const inputEl = document.getElementById(`commit-${itemId}`);
  const quantity = parseInt(inputEl.value);
  if (isNaN(quantity) || quantity < 0) {
    alert("Please enter a valid number for commitment.");
    return;
  }
  const { data: itemData, error: itemError } = await supabase
    .from('event_items')
    .select('quantity_required')
    .eq('id', itemId)
    .single();
  if (itemError || !itemData) {
    alert(`Error fetching item details: ${itemError?.message}`);
    return;
  }
  const requiredQuantity = itemData.quantity_required;
  const { data: commitmentsData } = await supabase
    .from('commitments')
    .select('quantity')
    .eq('item_id', itemId);
  const totalCommitted = commitmentsData?.reduce((sum, c) => sum + c.quantity, 0) || 0;
  const { data: myCommitData } = await supabase
    .from('commitments')
    .select('quantity')
    .eq('item_id', itemId)
    .eq('user_id', currentUser.id)
    .single();
  const myCurrent = myCommitData ? myCommitData.quantity : 0;
  const available = requiredQuantity - (totalCommitted - myCurrent);
  if (quantity > available) {
    alert(`Cannot commit ${quantity}. Only ${available} slots are available.`);
    return;
  }
  const { error } = await supabase
    .from('commitments')
    .upsert([{
      event_id: currentEvent.id,
      user_id: currentUser.id,
      item_id: itemId,
      quantity
    }], { onConflict: ['event_id', 'user_id', 'item_id'] });
  if (error) {
    alert(`Error committing item: ${error.message}`);
  } else {
    alert('Commitment successful!');
    loadEvent(currentEvent.slug);
  }
}
