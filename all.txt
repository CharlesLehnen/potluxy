-- Enable pgcrypto for UUID generation
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

---------------------------
-- 1. EVENTS TABLE
---------------------------
CREATE TABLE events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_name TEXT NOT NULL,
    slug TEXT UNIQUE NOT NULL,
    created_by TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    event_description TEXT,
    event_date DATE,
    event_time TIME
);

---------------------------
-- 2. EVENT ITEMS TABLE
---------------------------
CREATE TABLE event_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
    item_name TEXT NOT NULL,
    quantity_required INT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    item_description TEXT,
    UNIQUE(event_id, item_name)
);

---------------------------
-- 3. EVENT_USERS TABLE (Event-Specific Profiles)
---------------------------
CREATE TABLE event_users (
    id UUID PRIMARY KEY,
    event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
    username TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    UNIQUE(event_id, username)
);

---------------------------
-- 4. COMMITMENTS TABLE (User Commitments)
---------------------------
CREATE TABLE commitments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
    user_id UUID REFERENCES event_users(id) ON DELETE SET NULL,
    item_id UUID NOT NULL REFERENCES event_items(id) ON DELETE CASCADE,
    quantity INT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    UNIQUE(event_id, user_id, item_id)
);

---------------------------
-- Enable Row Level Security (RLS)
---------------------------
ALTER TABLE events ENABLE ROW LEVEL SECURITY;
ALTER TABLE event_items ENABLE ROW LEVEL SECURITY;
ALTER TABLE event_users ENABLE ROW LEVEL SECURITY;
ALTER TABLE commitments ENABLE ROW LEVEL SECURITY;

---------------------------
-- RLS Policies
---------------------------

-- EVENTS: Public SELECT and INSERT are allowed.
CREATE POLICY "Public select on events" ON events FOR SELECT USING (true);
CREATE POLICY "Public insert on events" ON events FOR INSERT WITH CHECK (true);

-- Only allow UPDATE/DELETE if the current auth user is the creator.
CREATE POLICY "Event update" ON events FOR UPDATE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = events.id
        AND event_users.username = events.created_by
    )
);
CREATE POLICY "Event delete" ON events FOR DELETE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = events.id
        AND event_users.username = events.created_by
    )
);

-- EVENT ITEMS: Public SELECT; for INSERT/UPDATE/DELETE, only allow if current user is the event creator.
CREATE POLICY "Event items select" ON event_items FOR SELECT USING (
    EXISTS (SELECT 1 FROM events WHERE events.id = event_items.event_id)
);
CREATE POLICY "Event items insert" ON event_items FOR INSERT
WITH CHECK (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = event_items.event_id
        AND event_users.username = (SELECT created_by FROM events WHERE events.id = event_items.event_id)
    )
);
CREATE POLICY "Event items update" ON event_items FOR UPDATE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = event_items.event_id
        AND event_users.username = (SELECT created_by FROM events WHERE events.id = event_items.event_id)
    )
);
CREATE POLICY "Event items delete" ON event_items FOR DELETE
USING (
    EXISTS (
        SELECT 1 FROM event_users
        WHERE event_users.id = auth.uid()
        AND event_users.event_id = event_items.event_id
        AND event_users.username = (SELECT created_by FROM events WHERE events.id = event_items.event_id)
    )
);

CREATE POLICY "Event users select" ON event_users FOR SELECT
  USING (true);

CREATE POLICY "Event users insert" ON event_users FOR INSERT
WITH CHECK (id = auth.uid());

-- COMMITMENTS: Users may only access their own commitments.
CREATE POLICY "Commitments select" ON commitments FOR SELECT
  USING (true);

CREATE POLICY "Commitments insert" ON commitments FOR INSERT
WITH CHECK (user_id = auth.uid());

CREATE POLICY "Commitments update" ON commitments FOR UPDATE
USING (user_id = auth.uid());

CREATE POLICY "Commitments delete" ON commitments FOR DELETE
USING (user_id = auth.uid());

---------------------------
-- Additional Constraints
---------------------------

-- Prevent overcommitments on INSERT
CREATE POLICY "Prevent overcommitments on insert"
ON commitments
FOR INSERT
WITH CHECK (
    quantity <= (
        SELECT quantity_required - COALESCE((
            SELECT SUM(quantity)
            FROM commitments
            WHERE item_id = commitments.item_id
        ), 0)
        FROM event_items
        WHERE id = commitments.item_id
    )
);

-- Prevent overcommitments on UPDATE
CREATE POLICY "Prevent overcommitments on update"
ON commitments
FOR UPDATE
WITH CHECK (
    quantity <= (
        SELECT quantity_required - COALESCE((
            SELECT SUM(quantity)
            FROM commitments
            WHERE item_id = commitments.item_id
        ), 0)
        FROM event_items
        WHERE id = commitments.item_id
    )
);


-- Prevent reducing quantity below existing commitments
CREATE OR REPLACE FUNCTION prevent_reduce_below_commitment()
RETURNS TRIGGER AS $$
DECLARE
    total_committed INT;
BEGIN
    SELECT COALESCE(SUM(quantity), 0)
    INTO total_committed
    FROM commitments
    WHERE item_id = NEW.id;

    IF NEW.quantity_required < total_committed THEN
        RAISE EXCEPTION 'Cannot reduce required quantity below existing commitments (%).', total_committed;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger to enforce the above constraint
CREATE TRIGGER prevent_reduce_below_commitment_trigger
BEFORE UPDATE ON event_items
FOR EACH ROW
EXECUTE FUNCTION prevent_reduce_below_commitment();

async function createEvent(e) {
  e.preventDefault();
  const eventName = document.getElementById("eventName").value.trim();
  const eventDescription = document.getElementById("eventDescription").value.trim();
  const eventDate = document.getElementById("eventDate").value;
  const eventTime = document.getElementById("eventTime").value;
  const username = document.getElementById("username").value.trim();
  const password = document.getElementById("password").value.trim();
  if (!eventName || !username) {
    alert("Event name and username are required.");
    return;
  }
  const response = await fetch("/api/createEvent", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ eventName, eventDescription, eventDate, eventTime, username, password }),
  });
  const result = await response.json();
  if (result.error) {
    alert("Error: " + result.error);
    return;
  }
  alert(`Event created successfully! Share this link: ${location.origin}/#/event/${result.slug}`);
  currentUserToken = result.token;
  window.location.hash = `#/event/${result.slug}`;
  loadEvent(result.slug);
}

async function loadEvent(slug) {
  const response = await fetch(`/api/event?slug=${slug}`);
  const result = await response.json();
  if (result.error) {
    alert("Event not found.");
    renderCreateEventForm();
    return;
  }
  currentEvent = result.event;
  if (!currentUserToken) {
    renderLoginForm(slug);
  } else {
    renderEventDashboard();
  }
}

async function renderCommitmentForm() {
  const response = await fetch(`/api/event?slug=${currentEvent.slug}`);
  const { commitments } = await response.json();
  const commitmentTotals = {};
  const userCommitments = {};
  if (commitments) {
    commitments.forEach((commit) => {
      commitmentTotals[commit.item_id] = (commitmentTotals[commit.item_id] || 0) + commit.quantity;
      if (commit.user_id === currentUserToken) {
        userCommitments[commit.item_id] = commit.quantity;
      }
    });
  }
  return currentEvent.items.map((item) => {
    const totalCommitted = commitmentTotals[item.id] || 0;
    const yourCommitment = userCommitments[item.id] || 0;
    const remaining = item.quantity_required - totalCommitted;
    return `
      <div class="commitment-card">
        <strong style="text-align: center; display: block;">${item.item_name}</strong>
        <br><br>
        <div>Required: ${item.quantity_required}</div>
        <div>Total Committed: ${totalCommitted}</div>
        <div>Your Commitment: ${yourCommitment}</div>
        <div>
          <input type="number" id="commit-${item.id}" value="${yourCommitment}" min="0">
          Remaining: ${remaining}
          <button onclick="commitItem('${item.id}')">Commit</button>
        </div>
      </div>
    `;
  }).join("");
}

async function renderCommitmentBreakdown() {
  const response = await fetch(`/api/event?slug=${currentEvent.slug}`);
  const { commitments, users } = await response.json();
  const userLookup = {};
  if (users) {
    users.forEach((user) => {
      if (user && user.username) {
        userLookup[user.id] = user.username;
      }
    });
  }
  const breakdownByItem = {};
  if (commitments) {
    commitments.forEach((row) => {
      const itemId = row.item_id;
      const username = row.user_id ? (userLookup[row.user_id] || "Unknown") : "Unknown";
      if (!breakdownByItem[itemId]) breakdownByItem[itemId] = [];
      breakdownByItem[itemId].push(`${username}: ${row.quantity}`);
    });
  }
  let html = "<h4>Commitment Breakdown</h4>";
  currentEvent.items.forEach((item) => {
    if (breakdownByItem[item.id]) {
      html += `<p><strong>${item.item_name}:</strong> ${breakdownByItem[item.id].join(", ")}</p>`;
    }
  });
  return html;
}

// addItem/index.ts – Supabase Edge Function to add an item
import { serve } from "https://deno.land/std@0.152.0/http/server.ts";
import { createClient } from "https://cdn.jsdelivr.net/npm/@supabase/supabase-js/+esm";

const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!;
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

// Helper: Validate Authorization header and return user ID.
async function getUserId(req: Request): Promise<string | null> {
  const authHeader = req.headers.get("Authorization");
  if (authHeader && authHeader.startsWith("Bearer ")) {
    const token = authHeader.slice(7);
    const { data: { user }, error } = await supabase.auth.getUser(token);
    if (error || !user) return null;
    return user.id;
  }
  return null;
}

serve(async (req) => {

// commitItem/index.ts – Supabase Edge Function to commit to an item
import { serve } from "https://deno.land/std@0.152.0/http/server.ts";
import { createClient } from "https://cdn.jsdelivr.net/npm/@supabase/supabase-js/+esm";

const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!;
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

async function getUserId(req: Request): Promise<string | null> {
  const authHeader = req.headers.get("Authorization");
  if (authHeader && authHeader.startsWith("Bearer ")) {
    const token = authHeader.slice(7);
    const { data: { user }, error } = await supabase.auth.getUser(token);
    if (error || !user) return null;
    return user.id;
  }
  return null;
}

serve(async (req) => {
  if (req.method !== "POST") {
    return new Response(JSON.stringify({ error: "Method not allowed" }), { status: 405 });
  }
  const userId = await getUserId(req);
  if (!userId) {
    return new Response(JSON.stringify({ error: "Unauthorized" }), { status: 401 });
  }
  try {
    const body = await req.json();
    const { itemId, quantity, eventSlug } = body;
    if (isNaN(quantity) || quantity < 0) {
      return new Response(JSON.stringify({ error: "Invalid commitment quantity" }), { status: 400 });
    }
    const { data: eventData } = await supabase
      .from("events")
      .select("id, slug")
      .eq("slug", eventSlug)
      .maybeSingle();
    if (!eventData) {
      return new Response(JSON.stringify({ error: "Event not found" }), { status: 404 });
    }
    const { data: itemData, error: itemError } = await supabase
      .from("event_items")
      .select("quantity_required")
      .eq("id", itemId)
      .single();
    if (itemError || !itemData) {
      return new Response(JSON.stringify({ error: "Item not found" }), { status: 404 });
    }
    const requiredQuantity = itemData.quantity_required;
    const { data: commitmentsData } = await supabase
      .from("commitments")
      .select("quantity")
      .eq("item_id", itemId);
    const totalCommitted = commitmentsData?.reduce((sum, c) => sum + c.quantity, 0) || 0;
    const { data: myCommitData } = await supabase
      .from("commitments")
      .select("quantity")
      .eq("item_id", itemId)
      .eq("user_id", userId)
      .maybeSingle();
    const myCurrent = myCommitData ? myCommitData.quantity : 0;
    const available = requiredQuantity - (totalCommitted - myCurrent);
    if (quantity > available) {
      return new Response(
        JSON.stringify({ error: `Cannot commit ${quantity}. Only ${available} slots available.` }),
        { status: 400 }
      );
    }
    const { error } = await supabase
      .from("commitments")
      .upsert([{
        event_id: eventData.id,
        user_id: userId,
        item_id: itemId,
        quantity,
      }], { onConflict: ["event_id", "user_id", "item_id"] });
    if (error) {
      return new Response(JSON.stringify({ error: error.message }), { status: 400 });
    }
    return new Response(JSON.stringify({ message: "Commitment successful" }), { headers: { "Content-Type": "application/json" } });
  } catch (err: any) {
    return new Response(JSON.stringify({ error: err.message }), { status: 500 });
  }
});

// createEvent/index.ts – Supabase Edge Function for creating an event
import { serve } from "https://deno.land/std@0.152.0/http/server.ts";
import { createClient } from "https://cdn.jsdelivr.net/npm/@supabase/supabase-js/+esm";

// Get these from your deployment environment
const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!;
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

function generateSlug(eventName: string) {
  const uniqueId = Math.random().toString(36).substring(2, 10);
  return eventName.toLowerCase().replace(/\s+/g, "-") + "-" + uniqueId;
}

function constructEmail(username: string, slug: string) {
  return `${username}.${slug}@mailinator.com`;
}

serve(async (req) => {
  if (req.method !== "POST") {
    return new Response(JSON.stringify({ error: "Method not allowed" }), { status: 405 });
  }
  try {
    const body = await req.json();
    const { eventName, eventDescription, eventDate, eventTime, username, password } = body;
    if (!eventName || !username) {
      return new Response(JSON.stringify({ error: "Event name and username are required" }), { status: 400 });
    }
    const slug = generateSlug(eventName);
    const email = constructEmail(username, slug);

    // Create the auth user via the admin API (bypassing email confirmation)
    const { data: userData, error: userError } = await supabase.auth.admin.createUser({
      email,
      password: password || "no-password",
      email_confirm: true,
    });
    if (userError) {
      return new Response(JSON.stringify({ error: userError.message }), { status: 400 });
    }
    const newUser = userData.user;

    // Insert the event record.
    const { data: eventData, error: eventError } = await supabase
      .from("events")
      .insert([{
        event_name: eventName,
        slug,
        created_by: username,
        event_description: eventDescription || null,
        event_date: eventDate || null,
        event_time: eventTime || null,
      }])
      .select()
      .single();
    if (eventError) {
      return new Response(JSON.stringify({ error: eventError.message }), { status: 400 });
    }
    const eventId = eventData.id;

    // Insert the event-specific user profile.
    const { data: profileData, error: profileError } = await supabase
      .from("event_users")
      .insert([{ id: newUser.id, event_id: eventId, username }])
      .select()
      .single();
    if (profileError) {
      return new Response(JSON.stringify({ error: profileError.message }), { status: 400 });
    }
    return new Response(
      JSON.stringify({ message: "Event created successfully", slug, token: newUser.id, eventId }),
      { headers: { "Content-Type": "application/json" } }
    );
  } catch (err: any) {
    return new Response(JSON.stringify({ error: err.message }), { status: 500 });

// login/index.ts – Supabase Edge Function for event login
import { serve } from "https://deno.land/std@0.152.0/http/server.ts";
import { createClient } from "https://cdn.jsdelivr.net/npm/@supabase/supabase-js/+esm";

const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!;
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

function constructEmail(username: string, slug: string) {
  return `${username}.${slug}@mailinator.com`;
}

serve(async (req) => {
  if (req.method !== "POST") {
    return new Response(JSON.stringify({ error: "Method not allowed" }), { status: 405 });
  }
  try {
    const body = await req.json();
    const { username, password, slug } = body;
    if (!username || !slug) {
      return new Response(JSON.stringify({ error: "Username and slug are required" }), { status: 400 });
    }
    const email = constructEmail(username, slug);

    // Attempt to sign in.
    const { data: signInData, error: signInError } = await supabase.auth.signInWithPassword({
      email,
      password: password || "no-password",
    });
    if (signInError) {
      // If invalid credentials, attempt to create a new user.
      if (signInError.message.includes("Invalid login credentials")) {
        const { data: signUpData, error: signUpError } = await supabase.auth.admin.createUser({
          email,
          password: password || "no-password",
          email_confirm: true,
        });
        if (signUpError) {
          return new Response(JSON.stringify({ error: signUpError.message }), { status: 400 });
        }
      } else {
        return new Response(JSON.stringify({ error: signInError.message }), { status: 400 });
      }
    }
    const user = signInData.session ? signInData.session.user : null;
    if (!user) {
      return new Response(JSON.stringify({ error: "User not authenticated" }), { status: 401 });
    }
    // Look up the event ID using the slug.
    const { data: eventRec } = await supabase
      .from("events")
      .select("id")
      .eq("slug", slug)
      .maybeSingle();
    if (!eventRec) {
      return new Response(JSON.stringify({ error: "Event not found" }), { status: 404 });
    }
    // Check if an event-specific profile exists.
    let { data: profileData } = await supabase
      .from("event_users")
      .select("*")
      .eq("id", user.id)
      .eq("event_id", eventRec.id)
      .maybeSingle();
    if (!profileData) {
      const { data: newProfile, error: newProfileError } = await supabase
        .from("event_users")
        .insert([{ id: user.id, event_id: eventRec.id, username }])
        .select()
        .single();
      if (newProfileError) {
        return new Response(JSON.stringify({ error: newProfileError.message }), { status: 400 });
      }
    }
    return new Response(
      JSON.stringify({ message: "Login successful", token: user.id }),
      { headers: { "Content-Type": "application/json" } }
    );
  } catch (err: any) {
    return new Response(JSON.stringify({ error: err.message }), { status: 500 });
  }
});
