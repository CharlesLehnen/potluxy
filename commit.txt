download and install: https://nodejs.org/en

conda create -n potluxy-nextjs nodejs=18

conda activate potluxy-nextjs

npx create-next-app@latest potluxy --typescript --eslint

- **Tailwind CSS** → Yes 
- **Use `src/` directory** → Yes 
- **Use App Router** → Yes 
- **Use Turbopack for `next dev`** → Yes 
- **Customize Import Aliases** → No 

npm install -g npm@11.2.0

npm install @supabase/supabase-js

create .env.local in potluxy folder:
NEXT_PUBLIC_SUPABASE_URL="https://your-project.supabase.co"
SUPABASE_SERVICE_ROLE_KEY="your-service-role-key"  # NEVER expose this in frontend

add that to .gitignore

create lib/supabase.ts in potluxy folder:
// lib/supabase.ts
import { createClient } from "@supabase/supabase-js";
const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY!;
export const supabase = createClient(supabaseUrl, serviceRoleKey);
